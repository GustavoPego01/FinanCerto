function getCurrentMonthKey() {
  const now = new Date();

  return `${now.getFullYear()}-${String(
    now.getMonth() + 1
  ).padStart(2, "0")}`;
}

/*
  Converte diferentes formatos de data
  para um objeto Date válido.
*/

function parseTransactionDate(dateValue) {
  if (!dateValue) return null;

  /*
    Caso já seja um objeto Date.
  */

  if (dateValue instanceof Date) {
    return dateValue;
  }

  const value = String(dateValue).trim();

  /*
    Formato YYYY-MM-DD
    Exemplo:
    2026-08-29
  */

  if (/^\d{4}-\d{2}-\d{2}$/.test(value)) {
    const [year, month, day] = value.split("-");

    return new Date(
      Number(year),
      Number(month) - 1,
      Number(day)
    );
  }

  /*
    Formato DD/MM/YYYY
    Exemplo:
    29/08/2026
  */

  if (/^\d{2}\/\d{2}\/\d{4}$/.test(value)) {
    const [day, month, year] = value.split("/");

    return new Date(
      Number(year),
      Number(month) - 1,
      Number(day)
    );
  }

  /*
    Tenta interpretar outros formatos.
  */

  const parsedDate = new Date(value);

  if (!Number.isNaN(parsedDate.getTime())) {
    return parsedDate;
  }

  return null;
}

/*
  Retorna a data sempre no formato visual:
  DD/MM/YYYY
*/

function formatTransactionDate(dateValue) {
  if (!dateValue) return "";

  const value = String(dateValue).trim();

  /*
    Se já estiver no formato correto.
  */

  if (/^\d{2}\/\d{2}\/\d{4}$/.test(value)) {
    return value;
  }

  const date = parseTransactionDate(value);

  if (!date) {
    return value;
  }

  const day = String(
    date.getDate()
  ).padStart(2, "0");

  const month = String(
    date.getMonth() + 1
  ).padStart(2, "0");

  const year = date.getFullYear();

  return `${day}/${month}/${year}`;
}

function getMonthKeyFromDate(dateValue) {
  const date = parseTransactionDate(dateValue);

  if (!date) {
    return getCurrentMonthKey();
  }

  return `${date.getFullYear()}-${String(
    date.getMonth() + 1
  ).padStart(2, "0")}`;
}

function formatMonthLabel(monthKey) {
  const [year, month] = monthKey.split("-");

  const date = new Date(
    Number(year),
    Number(month) - 1,
    1
  );

  return date.toLocaleDateString("pt-BR", {
    month: "long",
    year: "numeric",
  });
}

function formatMonthTitle(monthKey) {
  const label = formatMonthLabel(monthKey);

  return (
    label.charAt(0).toUpperCase() +
    label.slice(1)
  );
}

function changeMonth(monthKey, amount) {
  const [year, month] = monthKey.split("-");

  const date = new Date(
    Number(year),
    Number(month) - 1 + amount,
    1
  );

  return `${date.getFullYear()}-${String(
    date.getMonth() + 1
  ).padStart(2, "0")}`;
}

function isFutureMonth(monthKey) {
  return monthKey > getCurrentMonthKey();
}

/*
  Retorna a data atual de São Paulo
  no formato DD/MM/YYYY.
*/

function getToday() {
  const now = new Date();

  const parts = new Intl.DateTimeFormat(
    "pt-BR",
    {
      timeZone: "America/Sao_Paulo",
      day: "2-digit",
      month: "2-digit",
      year: "numeric",
    }
  ).formatToParts(now);

  const day = parts.find(
    (part) => part.type === "day"
  )?.value;

  const month = parts.find(
    (part) => part.type === "month"
  )?.value;

  const year = parts.find(
    (part) => part.type === "year"
  )?.value;

  return `${day}/${month}/${year}`;
}

/*
  Converte uma transação para garantir
  compatibilidade com os formatos existentes.
*/

function normalizeTransaction(transaction) {
  return {
    ...transaction,
    amount: Number(transaction.amount) || 0,
    date: formatTransactionDate(
      transaction.date
    ),
  };
}

/*
  Ordena transações da mais recente
  para a mais antiga.
*/

function sortTransactionsByDate(transactions) {
  return [...transactions].sort((a, b) => {
    const dateA = parseTransactionDate(a.date);
    const dateB = parseTransactionDate(b.date);

    const timeA = dateA
      ? dateA.getTime()
      : 0;

    const timeB = dateB
      ? dateB.getTime()
      : 0;

    return timeB - timeA;
  });
}

/* =========================================================

function formatCurrency(value) {
  return new Intl.NumberFormat("pt-BR", {
    style: "currency",
    currency: "BRL",
  }).format(Number(value) || 0);
}

function formatInputValue(value) {
  if (!value) return "";

  const numbers = value.replace(/\D/g, "");

  if (!numbers) return "";

  return (Number(numbers) / 100)
    .toFixed(2)
    .replace(".", ",");
}

function parseCurrency(value) {
  if (!value) return 0;

  const normalized = value
    .replace(/\./g, "")
    .replace(",", ".")
    .replace(/[^\d.-]/g, "");

  return Number(normalized) || 0;
}

      parseCurrency(amount);

    if (
      !numericAmount ||
      numericAmount <= 0
    ) {
      setMessage(
        "Digite um valor válido."
      );

      return;
    }

    if (!title.trim()) {
      setMessage(
        "Digite uma descrição."
      );

      return;
    }

    try {
      /*
        Pega usuário atualmente logado.
      */

      const {
        data: authData,
        error: authError,
      } =
        await supabase.auth.getUser();

      const currentUser =
        authData?.user;

      if (authError) {
        console.error(
          "ERRO AO PEGAR USUÁRIO:",
          authError
        );

        setMessage(
          "Erro ao identificar usuário."
        );

        return;
      }

      if (!currentUser) {
        setMessage(
          "Usuário não está logado."
        );

        return;
      }

      /*
        Cria nova transação.
      */

      console.log("DATA DE HOJE:", getToday());
      console.log("MÊS SELECIONADO:", selectedMonth);

      const newTransaction = {
        id: crypto.randomUUID(),

        user_id:
          currentUser.id,

        type:
          transactionType,

        title:
          title.trim(),

        category,

        amount:
          numericAmount,

        date:
          getToday(),
      };

      console.log(
        "ENVIANDO PARA SUPABASE:",
        newTransaction
      );

      /*
        Salva no Supabase.
      */

      const {
        data,
        error,
      } = await supabase
        .from("transactions")
        .insert([
          newTransaction,
        ])
        .select()
        .single();

      if (error) {
        console.error(
          "ERRO AO SALVAR NO SUPABASE:",
          error
        );

        setMessage(
          `Erro ao salvar: ${error.message}`
        );

        return;
      }

      console.log(
        "TRANSAÇÃO SALVA:",
        data
      );

      /*
        Normaliza dados recebidos.
      */

      const normalizedTransaction =
        normalizeTransaction(data);

      /*
        Atualiza tela.
      */

      setTransactions(
        (current) =>
          sortTransactionsByDate([
            normalizedTransaction,
            ...current,
          ])
      );

      /*
        Vai para o mês atual.
      */

      setSelectedMonth(
        getCurrentMonthKey()
      );

      closeModal();

      setPage("lancamentos");
    } catch (error) {
      console.error(
        "ERRO:",
        error
      );

      setMessage(
        "Ocorreu um erro ao salvar a transação."
      );
    }
  }

  /* =======================================================
     EXCLUIR TRANSAÇÃO
  ======================================================= */

  async function deleteTransaction(
    id
  ) {
    const confirmed =
      window.confirm(
        "Deseja realmente excluir este lançamento?"
      );

    if (!confirmed) return;

    try {
      /*
        Pega usuário logado.
      */

      const {
        data: authData,
        error: authError,
      } =
        await supabase.auth.getUser();

      const currentUser =
        authData?.user;

      if (
        authError ||
        !currentUser
      ) {
        console.error(
          "ERRO AO IDENTIFICAR USUÁRIO:",
          authError
        );

        return;
      }

      /*
        Exclui no Supabase.
      */

      const { error } =
        await supabase
          .from("transactions")
          .delete()
          .eq("id", id)
          .eq(
            "user_id",
            currentUser.id
          );

      if (error) {
        console.error(
          "ERRO AO EXCLUIR TRANSAÇÃO:",
          error
        );

        return;
      }

      /*
        Remove da tela
        somente após confirmação.
      */

      setTransactions(
        (current) =>
          current.filter(
            (transaction) =>
              transaction.id !== id
          )
      );

      console.log(
        "TRANSAÇÃO EXCLUÍDA COM SUCESSO:",
        id
      );
    } catch (error) {
      console.error(
        "ERRO AO EXCLUIR LANÇAMENTO:",
        error
      );
    }
  }

                      </option>
                    )
                  )}
                </select>
              </label>

              {/* MENSAGEM */}

              {message && (
                <p className="form-message">
                  {message}
                </p>
              )}

              {/* SALVAR */}

              <button
                className={`save-button ${transactionType ===
                  "income"
                  ? "income"
                  : "expense"
                  }`}
                type="submit"
              >
                Salvar{" "}
                {transactionType ===
                  "income"
                  ? "entrada"
                  : "gasto"}
              </button>
            </form>
          </div>
        </div>
      )}

      {/* ===============================================
          MENU MOBILE
      =============================================== */}

      {menuOpen && (
        <div
          className="mobile-menu-overlay"
          onClick={() =>
            setMenuOpen(false)
          }
        >
          <div
            className="mobile-menu"
            onClick={(
              event
            ) =>
              event.stopPropagation()
            }
          >
            <button
              onClick={() => {
                setPage("inicio");
                setMenuOpen(false);
              }}
            >
              Início
            </button>

            <button
              onClick={() => {
                setPage(
                  "lancamentos"
                );

                setMenuOpen(false);
              }}
            >
              Lançamentos
            </button>

            <button
              onClick={() => {
                setPage(
                  "relatorios"
                );

                setMenuOpen(false);
              }}
            >
              Relatórios
            </button>

            <button
              onClick={() => {
                setPage("perfil");

                setMenuOpen(false);
              }}
            >
              Perfil
            </button>
          </div>
        </div>
      )}
    </div>
  );
}

========================================================= */

function TransactionItem({
  transaction,
  showDelete,
  onDelete,
}) {
  const isIncome =
    transaction.type === "income";

  return (
    <div className="transaction-item">
      <div
        className={`transaction-icon ${isIncome
          ? "income"
          : "expense"
          }`}
      >
        {isIncome
          ? "↗"
          : "↘"}
      </div>

      <div className="transaction-info">
        <strong>
          {transaction.title}
        </strong>

        <span>
          {
            transaction.category ||
            "Outros"
          }{" "}
          •{" "}
          {formatTransactionDate(
            transaction.date
          )}
        </span>
      </div>

      <div className="transaction-value">
        <strong
          className={
            isIncome
              ? "green"
              : "red"
          }
        >
          {isIncome
            ? "+"
            : "-"}{" "}
          {formatCurrency(
            transaction.amount
          )}
        </strong>

        {showDelete && (
          <button
            className="delete-button"
            onClick={onDelete}
            title="Excluir lançamento"
            type="button"
          >
            🗑
          </button>
        )}
      </div>
    </div>
  );
}
